```tsx

  <PopoverTrigger />
  <PopoverAnchor />
  <PopoverPositioner>
    <PopoverBackdrop />
    <PopoverContent>
      <PopoverArrow />
    </PopoverContent>
  </PopoverPositioner>
</PopoverRoot>
```
