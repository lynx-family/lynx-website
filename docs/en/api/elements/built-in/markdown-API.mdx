## Properties

The frontend can set the corresponding properties on the component to define the runtime behavior of the component, as shown below.

```tsx
<markdown content={content} />
```

### `content`

```ts
/**
 * markdown source content.
 */
content: string;
```

markdown content

### `content-id`

```ts
/**
 * mark the content when the event or UI method is triggered.
 */
'content-id'?: string;
```

Used as a unique identifier for `content`, for consumption in events.

### `markdown-style`

```ts
/**
 * set the style of markdown.
 * @defaultValue {}
 */
'markdown-style'?: object
```

markdown style, for specific formats please refer to [Stylesheets](#stylesheets)

### `animation-type`

```ts
/**
 * which animation to use.
 * none: no animation
 * typewriter: typewriter animation
 * @defaultValue 'none'
 */
'animation-type'?: 'none' | 'typewriter'
```

markdown animation style, optional typewriter effect

### `animation-velocity`

```ts
/**
 * velocity of animation.
 * when animation-type is typewriter, it's the number of characters typed per second.
 * @defaultValue 1
 */
'animation-velocity'?:number;
```

Speed of markdown typewriter effect

### `text-maxline`

```ts
/**
 * maximum number of text lines.
 * a table row is counted as a line
 */
'text-maxline'?: number;
```

maximum number of markdown text lines

### `overflow`

```ts
overflow?: 'ellipsis' | 'clip';
```

Choose an ellipsis or clipping when text overflows. Defaults to `ellipsis`.

### `initial-animation-step`

```ts
/**
 * animation will start at the given step.
 * @defaultValue 0
 */
'initial-animation-step'?: number
```

Specifies the start position of markdown animation

### `content-complete`

```ts
/**
 * whether to hide the typewriter cursor.
 * true: typewriter cursor will be hidden and drawEnd event will be triggered after the animation is complete.
 * false: typewriter cursor will be shown and drawEnd event will not be triggered.
 * @defaultValue true
 */
'content-complete'?: boolean;
```

Indicates if the current `content` is complete. If `true`, the cursor will be hidden after the animation ends, otherwise, the cursor remains visible

### `typewriter-dynamic-height`

```ts
/**
 * whether markdown height is equal to the content height typed by typewriter
 * @defaultValue false
 */
'typewriter-dynamic-height'?: boolean
```

Configures typewriter effect to dynamically change element height

### `text-selection`

```ts
/**
 * whether markdown content is selectable.
 * @defaultValue false
 */
'text-selection'?: boolean
```

Configures whether the markdown content is selectable

### `text-mark-attachments`

Used for complex highlighting: overlapping segments, line-wrapping, combined with underlines or strokes.

```ts
/**
 * define text mark attachments for highlighting or underlining character ranges.
 * attachments are drawn per line while preserving gradient continuity across lines.
 * @defaultValue []
 */
'text-mark-attachments'?: MarkAttachment[];

interface LineStyle {
  lineType: 'none' | 'solid' | 'dashed';
  color: string; // supports linear-gradient
  width: number; // px
  elementSize?: number; // dashed segment length
  emptySize?: number; // dashed gap length
}

interface MarkStyle {
  radius?: number; // corner radius
  left?: number;
  right?: number;
  top?: number;
  bottom?: number;
  color?: string; // fill color or gradient
  borderLeft?: LineStyle;
  borderTop?: LineStyle;
  borderRight?: LineStyle;
  borderBottom?: LineStyle;
}

interface MarkAttachment {
  startIndex: number;
  endIndex: number; // exclusive
  indexType: 'char' | 'source';
  clickable?: boolean;
  id?: string;
  layer?: 'background' | 'foreground';
  style: MarkStyle;
}
```

### `content-range`

```ts
/**
 * the range of content to be parsed.
 * @defaultValue [0, int32_max]
 */
'content-range'?: number[];
```

Configures the parsing range for markdown content

### `markdown-effect`

```ts
/**
 * additional effects
 * @defaultValue {}
 */
'markdown-effect'?: object
```

Configure additional rendering effects. A text mask uses `{ type: 'text-mask', color, rangeStart, rangeEnd }`; see [Text Fade Effect](#text-fade-effect).

### `typewriter-height-transition-duration`

```ts
/**
 * transition duration when height extended by typewriter animation.
 * @defaultValue 0
 */
'typewriter-height-transition-duration'?: number
```

Set the transition duration in seconds when typewriter content changes height. The default is `0`. Use with `typewriter-dynamic-height`.

### `enable-gif`

```ts
/**
 * whether to enable gif animation.
 * @defaultValue false
 */
'enable-gif'?: boolean
```

Configure whether to enable GIF animation on iOS. Android supports GIF animation by default.

## Methods

The frontend can use the [SelectorQuery](/api/lynx-api/nodes-ref/nodes-ref-invoke.md) API to execute the component's methods, as shown below:

```ts title=index.js
lynx
  .createSelectorQuery() // create SelectorQuery
  .select('#markdown') // specify the target node's selector
  .invoke({
    // specify the operation for the target node
    method: 'pauseAnimation',
    success: function (res) {
      console.log(res);
    },
    fail: function (res) {
      console.log(res.code, res.data);
    },
  })
  .exec(); // execute query
```
